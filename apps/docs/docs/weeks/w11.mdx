<DemoBox title="命令白名单沙箱适配器" goal="即使还没接 E2B，也禁止 Agent 直接 child_process 任意命令。">

```ts
const ALLOWED = new Set(['npm test', 'pnpm test', 'python -m pytest']);

export function assertSafeCommand(cmd: string) {
  if (!ALLOWED.has(cmd)) {
    throw new Error(`command_not_allowed: ${cmd}`);
  }
}

export async function runInSandbox(cmd: string, tenant: string) {
  assertSafeCommand(cmd);
  return { tenant, cmd, where: 'sandbox', exitCode: 0 };
}
```

验收：`rm -rf /` 或 `cat ~/.env` 抛错。真正接 E2B 时保持同一函数签名。
