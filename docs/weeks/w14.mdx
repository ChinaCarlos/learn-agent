<DemoBox title="发布清单与自动回滚条件" goal="生产化不是多写 YAML，而是用代码表达何时认为自己失败。">

```ts
type Slo = { errorRate: number; p95ms: number; usdPerRun: number };

export function shouldRollback(now: Slo, budget: Slo) {
  if (now.errorRate > budget.errorRate) return 'error_rate';
  if (now.p95ms > budget.p95ms) return 'latency';
  if (now.usdPerRun > budget.usdPerRun) return 'cost';
  return null;
}

export const STAGING_BUDGET: Slo = { errorRate: 0.05, p95ms: 8000, usdPerRun: 1.5 };
```

验收：把 errorRate 调到 0.2 必须返回 `error_rate`。把这个函数接到你的 canary 脚本。
