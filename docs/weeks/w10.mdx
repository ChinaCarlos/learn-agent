<DemoBox title="并发槽位 + 成本拒绝" goal="Harness 最容易自测的两件事：同时跑几个、钱够不够。">

```ts
export class Harness {
  constructor(
    private readonly concurrency: number,
    private running = 0,
    private readonly dailyUsd: number,
    private spent = 0,
  ) {}

  admit(estimatedUsd: number) {
    if (this.spent + estimatedUsd > this.dailyUsd) {
      return { ok: false as const, code: 429, error: 'cost_cap' };
    }
    if (this.running >= this.concurrency) {
      return { ok: false as const, code: 202, error: 'queued' };
    }
    this.running += 1;
    this.spent += estimatedUsd;
    return { ok: true as const, code: 200 };
  }

  finish() {
    this.running = Math.max(0, this.running - 1);
  }
}
```

验收：concurrency=1 时第二次 `admit` 得 queued；日预算用尽得 429。
