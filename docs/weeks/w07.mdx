<DemoBox title="最小 Supervisor 路由" goal="演示「找同事」而不是「再调一个函数」。">

```ts
type AgentId = 'analyst' | 'specifier' | 'developer';

export function nextAgent(task: { hasAnalysis: boolean; hasStories: boolean }): AgentId {
  if (!task.hasAnalysis) return 'analyst';
  if (!task.hasStories) return 'specifier';
  return 'developer';
}

export async function handoff(from: AgentId, to: AgentId, payload: unknown) {
  return {
    protocol: 'A2A',
    binding: 'jsonrpc',
    from,
    to,
    payload,
  };
}
```

验收：缺 analysis 时不得跳到 developer。把 `handoff` 的返回对象打进日志，这就是「协作 trace」的胚胎。
