<DemoBox title="W3C Trace Context 解析" goal="先把传播做对，再谈用哪家观测产品。">

```ts
export function parseTraceparent(header?: string) {
  if (!header) return null;
  const [version, traceId, spanId, flags] = header.split('-');
  if (version !== '00' || traceId.length !== 32 || spanId.length !== 16) {
    return null;
  }
  return { version, traceId, spanId, flags };
}

export function childTraceparent(parent: { traceId: string; flags: string }, childSpanId: string) {
  return `00-${parent.traceId}-${childSpanId}-${parent.flags}`;
}
```

验收：从 UI 复制一个 header，API 与 worker 日志里 `traceId` 必须相同。
