<DemoBox title="无状态头路由" goal="演示方法名走头、请求自包含。这是新规范的最小骨架。">

```ts
import type { IncomingHttpHeaders } from 'node:http';

type McpCall = { method: string; name: string; tenant: string; args: unknown };

export function parseMcpHeaders(headers: IncomingHttpHeaders, body: unknown): McpCall {
  const method = headers['mcp-method'];
  const name = headers['mcp-name'];
  const tenant = headers['x-tenant-id'];
  if (typeof method !== 'string' || typeof name !== 'string' || typeof tenant !== 'string') {
    throw new Error('missing_mcp_headers');
  }
  return { method, name, tenant, args: body };
}

export async function handleMcp(call: McpCall) {
  if (call.method === 'tools/call' && call.name === 'repo.search') {
    return { files: ['README.md'], tenant: call.tenant };
  }
  throw new Error('unknown_method');
}
```

验收：缺头抛错；带齐头返回文件列表。把每次 call JSON.append 到 `audit.ndjson`。
